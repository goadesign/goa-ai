// Package evalconsumer implements and runs the hook interface generated from
// the fixture design, proving the complete downstream consumption contract.
package evalconsumer

import (
	"context"
	"fmt"
	"testing"

	genevals "example.com/evalconsumer/gen/evals/assistant_quality"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
	aieval "goa.design/goa-ai/eval"
	"goa.design/goa-ai/runtime/agent/tools"
)

type hooks struct{}

type judge struct{}

func (judge) Judge(_ context.Context, assertions []aieval.Assertion) ([]aieval.Judgment, error) {
	judgments := make([]aieval.Judgment, len(assertions))
	for index, assertion := range assertions {
		var label aieval.Label
		switch {
		case assertion.Output == "The pump is running." &&
			assertion.Claim == "The pump is running.":
			label = aieval.Entailed
		case assertion.Output == "The pump is stopped." &&
			assertion.Claim == "The pump is running.":
			label = aieval.Contradicted
		case assertion.Output == "The valve is open." &&
			assertion.Claim == "The pump is running.":
			label = aieval.NotAddressed
		case assertion.Output == "Two readings at the same time disagree about whether the pump is running." &&
			assertion.Claim == "The pump is running.":
			label = aieval.Indeterminate
		case assertion.Output == "Record summary complete." &&
			assertion.Claim == "The response includes a record summary.":
			label = aieval.Entailed
		default:
			return nil, fmt.Errorf("fixture judge received unknown semantic assertion %q", assertion.ClaimID)
		}
		judgments[index] = aieval.Judgment{
			ClaimID:   assertion.ClaimID,
			Label:     label,
			Rationale: "Deterministic fixture judgment.",
		}
	}
	return judgments, nil
}

func (hooks) RecordSummary(context.Context, *genevals.QueryEvalInput) (aieval.Result, error) {
	return aieval.Result{
		Checks: []aieval.Check{{
			Name:   "record_summary",
			Passed: true,
		}},
		Claims: []aieval.Claim{{
			ID:   "response_includes_summary",
			Text: "The response includes a record summary.",
		}},
		Output: "Record summary complete.",
	}, nil
}

func (hooks) SavedQueryReplay(context.Context, *genevals.SavedQueryEvalInput) (aieval.Result, error) {
	return aieval.Result{
		Checks: []aieval.Check{{
			Name:   "saved_query_replay",
			Passed: true,
		}},
	}, nil
}

func TestGeneratedSuite(t *testing.T) {
	runner, err := aieval.NewRunner(judge{}, aieval.RunnerConfig{MaxConcurrency: 2})
	require.NoError(t, err)

	suite, err := genevals.New(hooks{}, genevals.Inputs{
		RecordSummary: &genevals.QueryEvalInput{
			Query: "Summarize the matching records.",
		},
		SavedQueryReplay: &genevals.SavedQueryEvalInput{
			SavedQueryID: "00000000-0000-4000-8000-000000000004",
		},
	})
	require.NoError(t, err)
	report, err := runner.RunScenarios(
		context.Background(),
		suite,
		"record_summary",
		"saved_query_replay",
	)

	require.NoError(t, err)
	assert.True(t, report.Passed)
	assert.Len(t, report.Scenarios, 2)
	assert.NotNil(t, genevals.MustToolContract(tools.Ident("assistant_tools.answer")))
	assert.NotNil(t, genevals.MustToolContract(tools.Ident("records.lookup")))
}
